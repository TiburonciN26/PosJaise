// Reconstruye el estado V1 (antes de las correcciones V2) de los archivos que V2 modificó, para el rollback V2→V1.
//  · main.jsx: copia «antes» de la Fase 8 (V1 no lo tocó; hash en antes/SHA256.txt).
//  · vite.config.js y paginasCliente.js: copia «antes» + parche V1 (aplicado aparte con git apply en un árbol temporal).
//  · el resto: deshaciendo exactamente las sustituciones de texto de V2.
const fs = require('fs'), path = require('path')
const [repo, salida] = process.argv.slice(2)
const leer = (f) => fs.readFileSync(path.join(repo, f), 'utf8')
const escribir = (f, t) => { fs.mkdirSync(path.dirname(path.join(salida, f)), { recursive: true }); fs.writeFileSync(path.join(salida, f), t) }
function deshacer(f, pares) {
  let t = leer(f)
  for (const [v2, v1] of pares) {
    if (!t.includes(v2)) throw new Error(`${f}: no se encontró el texto V2: ${v2.slice(0, 80)}`)
    t = t.split(v2).join(v1)
  }
  return t
}
