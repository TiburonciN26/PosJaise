async function caso(nombre, fn) {
  if (SOLO && !nombre.includes(SOLO)) return
  const t0 = Date.now()
  try { const detalle = await fn(); casos.push({ nombre, ok: true, ms: Date.now() - t0, detalle }); console.log(`OK    ${nombre}`) } catch (e) {
    const d = String(e.message).split('\n').filter(Boolean).slice(0, 6).join(' | ').slice(0, 900)
    casos.push({ nombre, ok: false, error: d, ms: Date.now() - t0 }); console.log(`FALLA ${nombre}\n      ${d}`)
  }
}

// Pantalla cargada: su contenedor existe, sin «Cargando...» ni aviso de fallo, con contenido real.
const pantallaLista = (page, ruta, timeout = 20000) => page.waitForFunction((r) => {
  const c = document.querySelector(`[data-ruta="${r}"]`)
  return c && !c.querySelector('[data-carga-fallida]') && !/^\s*Cargando\.\.\.\s*$/.test(c.innerText) && c.innerText.trim().length > 40
}, ruta, { timeout })
const errorGlobal = (page) => page.getByText('Algo salió mal', { exact: true }).count()
// Abre una pestaña del POS y espera su desenlace: lista, aviso local de fallo o límite global («Algo salió mal»).
async function abrirPantalla(page, ruta) {
  await irA(page, ruta)
  await page.waitForFunction((r) => {
    if (document.body.innerText.includes('Algo salió mal')) return true
    const c = document.querySelector(`[data-ruta="${r}"]`)
    return c && (c.querySelector('[data-carga-fallida]') || (!/^\s*Cargando\.\.\.\s*$/.test(c.innerText) && c.innerText.trim().length > 40))
  }, ruta, { timeout: 20000 })
  return { errorGlobal: await errorGlobal(page), avisoFallo: await page.locator(`[data-ruta="${ruta}"] [data-carga-fallida]`).count(), carrito: await enCarrito(page) }
}
