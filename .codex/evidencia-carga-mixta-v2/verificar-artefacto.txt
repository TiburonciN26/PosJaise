const sources=Object.entries(mf.functions).map(([f,h])=>({archivo:f,shaManifiesto:h,shaCommit:hash(execFileSync('git',['show',mf.commit+':'+f],{cwd:'C:/WedJaiseReact'})),incluidoEnPaquete:fs.existsSync(path.join(base,f))}))
