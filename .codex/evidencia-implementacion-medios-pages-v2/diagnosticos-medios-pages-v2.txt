import { manejarMedios } from 'file:///C:/WedJaiseReact/cloudflare/medios-pages/entrega.js'

import assert from 'node:assert/strict'

const ruta = 'https://staging.pages.dev/medios/fotos-productos/00000000-0000-4000-8000-000000000001/g.webp'
const resultados = []
const cabecerasSeguras = (r) => r.headers.get('X-Content-Type-Options') === 'nosniff' && r.headers.get('X-Robots-Tag') === 'noindex'
let fallaCuerpo = true
let puts = 0
const pendientes = []
const env = { MEDIOS: { get: async () => ({ httpEtag: '"ficticio"', arrayBuffer: async () => { if (fallaCuerpo) throw new Error('fallo ficticio leyendo cuerpo'); return new Uint8Array([1, 2, 3]).buffer } }) } }
const cache = { match: async () => undefined, put: async () => { puts++ } }
const ctx = { waitUntil: (p) => pendientes.push(p) }
const error = await manejarMedios(new Request(ruta), env, ctx, { cache })
await Promise.all(pendientes.splice(0))
assert.equal(error.status, 503)
assert.equal(error.headers.get('Cache-Control'), 'no-store')
assert.ok(cabecerasSeguras(error))
assert.equal(await error.text(), 'No disponible')
assert.equal(puts, 0)
resultados.push({ caso: 'PFM-01 GET original: get resuelve y cuerpo rechaza', status: error.status, noStore: true, cabecerasSeguras: true, puts })
fallaCuerpo = false
const sano = await manejarMedios(new Request(ruta), env, ctx, { cache })
await Promise.all(pendientes.splice(0))
assert.equal(sano.status, 200)
assert.deepEqual([...new Uint8Array(await sano.arrayBuffer())], [1, 2, 3])
assert.equal(puts, 1)
resultados.push({ caso: 'Recuperacion GET tras fallo', status: sano.status, puts })

for (const conHead of [true, false]) {
  let heads = 0
  let gets = 0
  let cuerpos = 0
  let putsHead = 0
  const binding = { get: async () => { gets++; return { httpEtag: '"ficticio"', arrayBuffer: async () => { cuerpos++; throw new Error('no debe consumirse') } } } }
  if (conHead) binding.head = async () => { heads++; return { httpEtag: '"ficticio"' } }
  const r = await manejarMedios(new Request(ruta, { method: 'HEAD' }), { MEDIOS: binding }, ctx, { cache: { match: async () => undefined, put: async () => { putsHead++ } } })
  assert.equal(r.status, 200)
  assert.equal((await r.arrayBuffer()).byteLength, 0)
  assert.equal(cuerpos, 0)
  assert.equal(putsHead, 0)
  assert.equal(heads, conHead ? 1 : 0)
  assert.equal(gets, conHead ? 0 : 1)
  resultados.push({ caso: 'HEAD metadatos y cuerpo que fallaria', conHead, status: r.status, heads, gets, cuerpos, putsHead })
}

for (const tipo of ['binding ausente', 'objeto ausente', 'fallo metadatos', 'ruta invalida']) {
  const binding = { get: async () => null, head: async () => { if (tipo === 'fallo metadatos') throw new Error('ficticio'); return null } }
  const r = await manejarMedios(new Request(tipo === 'ruta invalida' ? 'https://staging.pages.dev/medios' : ruta, { method: 'HEAD' }), tipo === 'binding ausente' ? {} : { MEDIOS: binding }, ctx, { cache })
  assert.equal(r.status, tipo === 'fallo metadatos' ? 503 : 404)
  assert.equal((await r.arrayBuffer()).byteLength, 0)
  assert.equal(r.headers.get('Cache-Control'), 'no-store')
  assert.ok(cabecerasSeguras(r))
  resultados.push({ caso: `HEAD ${tipo}`, status: r.status, bytesRespuesta: 0, noStore: true })
}
