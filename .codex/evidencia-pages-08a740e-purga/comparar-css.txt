import fs from 'node:fs';import crypto from 'node:crypto';
const destinos=['https://90303ee8.pos-jaise.pages.dev','https://feat-cloudflare-pages.pos-jaise.pages.dev'];const evidencia=[];
for(const origin of destinos){const r=await fetch(origin+'/',{redirect:'error'});if(!r.ok)throw new Error('HTML fallido');const html=await r.text();const rutas=[...html.matchAll(/href="([^"]+\.css)"/g)].map(m=>m[1]);if(rutas.length!==1)throw new Error('CSS no identificable');const u=new URL(rutas[0],origin+'/');if(u.origin!==origin)throw new Error('CSS fuera de origen');const c=await fetch(u,{redirect:'error'});if(!c.ok)throw new Error('CSS fallido');const b=Buffer.from(await c.arrayBuffer());evidencia.push({origin,ruta:rutas[0],bytes:b.length,sha256:crypto.createHash('sha256').update(b).digest('hex')});}
const result={fecha:new Date().toISOString(),evidencia,identicos:evidencia[0].sha256===evidencia[1].sha256};fs.writeFileSync(new URL('./css-comparado.json',import.meta.url),JSON.stringify(result,null,2),{flag:'wx'});console.log(JSON.stringify(result,null,2));process.exitCode=result.identicos?0:1;
